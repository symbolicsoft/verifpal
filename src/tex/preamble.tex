\newcommand{\vpwire}[8]{%
  \ifnum#2>-1\relax\def\vpfrom{#2}\else\def\vpfrom{#1}\fi
  \pgfmathsetlength{\vpspan}{max(1.2cm,abs(#3-\vpfrom)*\vplanesep-0.7cm)}%
  \setbox\vpmeasure\hbox{\footnotesize#7}%
  \ifdim\wd\vpmeasure>\vpspan
    \setbox\vpmeasure\hbox{\parbox{\vpspan}{\centering\footnotesize#7}}%
    \tikzset{vpwrap/.style={text width=\vpspan,align=center,anchor=south,
      fill=none,yshift=1pt}}%
    \pgfmathsetmacro{\vprowh}{(\ht\vpmeasure+\dp\vpmeasure)/1cm+0.62}%
    \pgfmathsetmacro{\vparrowy}{-\vprowh+0.3}%
  \else
    \tikzset{vpwrap/.style={}}%
    \def\vprowh{\vprowsep}%
    \pgfmathsetmacro{\vparrowy}{-\vprowh/2}%
  \fi
  \vprow{\vprowh}{%
    \let\vpy\vparrowy
    \ifnum#2>-1\relax\draw[vpmuted] (#1,\vpy) -- (#2,\vpy);\fi
    \draw[vparrow,#4] (\vpfrom,\vpy) -- (#3,\vpy) node[#8,midway,vpwrap]{#7};
    \vpstep{#5}{#6}}}

\newcommand{\vpslice}[1]{%
  \global\advance\vplineat by 1
  \ifnum\vplineat=1 \vpfirstslicetrue\else\vpfirstslicefalse\fi
  \ifnum\vplineat=\vplinecount\vplastslicetrue\else\vplastslicefalse\fi
  \setbox\vpmeasure\hbox{\parbox[t]{\vpthisw}{\scriptsize\raggedright\strut#1\strut}}%
  \pgfmathsetmacro{\vptop}{\ifvpfirstslice\vpboxgap+\vpboxpad\else0\fi}%
  \pgfmathsetmacro{\vpyt}{\ifvpfirstslice-\vpboxgap\else0\fi}%
  \pgfmathsetmacro{\vpyb}%
    {-(\vptop+(\ht\vpmeasure+\dp\vpmeasure)/1cm+\ifvplastslice\vpboxpad\else0\fi)}%
  \vprow{0-(\vpyb)+\ifvplastslice\vpboxgap\else0\fi}{%
    \coordinate (vptl) at ($(\vpboxlane,\vpyt)+(-0.5\vpthisw-5pt,0)$);
    \coordinate (vptr) at ($(\vpboxlane,\vpyt)+(0.5\vpthisw+5pt,0)$);
    \coordinate (vpbl) at ($(\vpboxlane,\vpyb)+(-0.5\vpthisw-5pt,0)$);
    \coordinate (vpbr) at ($(\vpboxlane,\vpyb)+(0.5\vpthisw+5pt,0)$);
    \ifvpfirstslice
      \ifvplastslice
        \path[vpboxnow,rounded corners=1.5pt] (vptl) rectangle (vpbr);
      \else
        \path[vpboxnow,rounded corners=1.5pt] (vpbl) -- (vptl) -- (vptr) -- (vpbr);
      \fi
    \else
      \ifvplastslice
        \path[vpboxnow,rounded corners=1.5pt] (vptl) -- (vpbl) -- (vpbr) -- (vptr);
      \else
        \path[vpboxnow,draw=none] (vptl) rectangle (vpbr);
        \path[vpboxnow,fill=none] (vptl) -- (vpbl) (vptr) -- (vpbr);
      \fi
    \fi
    \node[vpboxtext,anchor=north west,inner sep=0pt]
      at ($(\vpboxlane,-\vptop)+(-0.5\vpthisw,0)$) {\usebox{\vpmeasure}};}}

\newcommand{\vpsplitloop}{%
  \vpfirst=\vpavail
  \advance\vpfirst by -\ht\vpplates
  \advance\vpfirst by -1cm
  \ifdim\vpfirst<3cm
    \vpfirst=\textheight
    \advance\vpfirst by -\ht\vpplates
  \fi
  \splittopskip=0pt
  \splitmaxdepth=\maxdimen
  \vbadness=10000
  \loop
    \vptmp=\vpfirst
    \advance\vptmp by -\vpreserve
    \ifdim\ht\vpall>\vptmp
      \ifdim\ht\vpall>\vpfirst\else\vpfirst=\vptmp\fi
      \setbox\vpchunk=\vsplit\vpall to \vpfirst
      \setbox\vpchunk=\vbox{\unvbox\vpchunk}%
    \else
      \setbox\vpchunk=\box\vpall
    \fi
    \vpemitchunk{\vpchunk}%
    \vpcontinuedtrue
    \ifvoid\vpall\vpmorefalse\else\vpmoretrue\fi
  \ifvpmore
    \penalty-10000
    \vpfirst=\textheight
    \advance\vpfirst by -\ht\vpplates
    \advance\vpfirst by -0.7cm
  \repeat}

\newcommand{\vpemit}{%
  \vptmp=\ht\vpall
  \advance\vptmp by \ht\vpplates
  \advance\vptmp by \vpreserve
  \ifdim\pagegoal=\maxdimen
    \vpavail=\textheight
  \else
    \vpavail=\pagegoal
    \advance\vpavail by -\pagetotal
  \fi
  \vpcontinuedfalse
  \vpsplitfalse
  \ifinner\else
    \ifdim\vptmp>\vpavail
      \ifdim\vptmp>\textheight
        \vpsplittrue
      \else
        \ifdim\vpavail>0.5\textheight\vpsplittrue\fi
      \fi
    \fi
  \fi
  \ifvpsplit\vpsplitloop\else\vpemitchunk{\vpall}\fi
  \nopagebreak}
